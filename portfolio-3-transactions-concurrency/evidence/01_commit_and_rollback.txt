========== A. Placing an order: four changes that must all happen together, then COMMIT ==========
SELECT product_name, stock_qty FROM products WHERE product_id = 2;
    product_name    | stock_qty 
--------------------+-----------
 Samsung Galaxy A15 |        12
(1 row)

BEGIN;
BEGIN
INSERT INTO orders (customer_id, shipping_address_id, order_status, ordered_at)
VALUES (1, 1, 'pending', timestamptz '2026-10-31 18:00') RETURNING order_id AS order_a 
INSERT 0 1
INSERT INTO order_items (order_id, product_id, quantity, unit_price) VALUES (600012, 2, 2, 245000.00);
INSERT 0 1
UPDATE products SET stock_qty = stock_qty - 2 WHERE product_id = 2;
UPDATE 1
INSERT INTO payments (order_id, amount, payment_method, payment_status, reference)
VALUES (600012, 490000.00, 'card', 'successful', 'DEMO-' || 600012);
INSERT 0 1
UPDATE orders SET order_status = 'paid' WHERE order_id = 600012;
UPDATE 1
COMMIT;
COMMIT
SELECT o.order_status, oi.quantity, p.payment_status, p.amount,
       (SELECT stock_qty FROM products WHERE product_id = 2) AS stock_left
FROM orders o JOIN order_items oi USING (order_id) JOIN payments p USING (order_id)
WHERE o.order_id = 600012;
